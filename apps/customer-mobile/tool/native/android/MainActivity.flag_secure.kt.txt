// Blocks screenshots, screen recording and the preview in the "recent apps" list on

// customers cannot take a screenshot for support either. Decide per deployment.
//

//   android/app/src/main/kotlin/<your/package/path>/MainActivity.kt
// with the class below (keep the `package ...` line that Flutter generated).

        )
    }
}
